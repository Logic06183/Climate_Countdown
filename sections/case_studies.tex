\subsection{Eastern Africa: Heat Waves and Vector-Borne Diseases}
\textbf{Case Study: Kenya and Tanzania (2024)}

\begin{itemize}
    \item \textbf{Context:}
    \begin{itemize}
        \item Record temperatures in 2023-2024
        \item Increased malaria transmission
        \item Drought impacts on food security
    \end{itemize}
    
    \item \textbf{Health Impacts:}
    \begin{itemize}
        \item 45\% increase in heat-related illnesses
        \item Extended malaria transmission seasons
        \item Rising malnutrition cases
    \end{itemize}
    
    \item \textbf{Adaptation Measures:}
    \begin{itemize}
        \item Early warning systems
        \item Vector control programs
        \item Community health worker networks
    \end{itemize}
\end{itemize}
